**4. Replace `apps/frontend/src/App.tsx`.**

This is the critical wiring file. It sets up:
- React Router with nested routes
- Auth hydration on app mount (attempt silent refresh)
- Route structure: public routes under AuthLayout, private routes under PrivateRoute -> AppLayout

```typescript
import { useEffect } from 'react';
import {
  createBrowserRouter,
  Navigate,
  RouterProvider,
} from 'react-router';

import { AuthLayout } from '@/components/layouts/auth-layout';
import { AppLayout } from '@/components/layouts/app-layout';
import { PrivateRoute } from '@/components/layouts/private-route';
import { api } from '@/lib/axios';
import { DashboardPage } from '@/pages/dashboard';
import { LoginPage } from '@/pages/login';
import { PlaceholderPage } from '@/pages/placeholder';
import { ProfilePage } from '@/pages/profile';
import { SignupPage } from '@/pages/signup';
import { useAuthStore } from '@/stores/auth-store';

const router = createBrowserRouter([
  {
    element: <AuthLayout />,
    children: [
      { path: '/login', element: <LoginPage /> },
      { path: '/signup', element: <SignupPage /> },
    ],
  },
  {
    element: <PrivateRoute />,
    children: [
      {
        element: <AppLayout />,
        children: [
          { path: '/dashboard', element: <DashboardPage /> },
          { path: '/profile', element: <ProfilePage /> },
          { path: '/courses', element: <PlaceholderPage title="Courses" /> },
          { path: '/assignments', element: <PlaceholderPage title="Assignments" /> },
          { path: '/gradebook', element: <PlaceholderPage title="Gradebook" /> },
        ],
      },
    ],
  },
  { path: '/', element: <Navigate to="/dashboard" replace /> },
  { path: '*', element: <Navigate to="/dashboard" replace /> },
]);

export const App = () => {
  const { setAuth, setHydrating, clearAuth } = useAuthStore();

  useEffect(() => {
    // Attempt silent refresh on app mount to restore session
    const hydrate = async (): Promise<void> => {
      try {
        const { data } = await api.post('/auth/refresh');
        setAuth(data.data.user, data.data.accessToken);
      } catch {
        // No valid refresh token — user needs to log in
        clearAuth();
      }
    };

    hydrate();
  }, [setAuth, setHydrating, clearAuth]);

  return <RouterProvider router={router} />;
};
```

IMPORTANT: The `hydrate` function calls `api.post('/auth/refresh')`. If the refresh token cookie exists and is valid, the backend returns new tokens and user data. The `setAuth` call sets `isHydrating: false` and `isAuthenticated: true`. If the refresh fails (no cookie or expired), `clearAuth()` sets `isHydrating: false` and `isAuthenticated: false`, which causes PrivateRoute to redirect to /login.

The `useEffect` runs once on mount. The silent refresh interceptor in axios.ts handles subsequent 401s during the session.

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| user input -> form | Untrusted user input in login, signup, and profile forms |
| ?next= query param | User-controlled redirect destination — must be validated |
| API response -> Zustand store | Backend response data stored in memory — trusted after server validation |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-18 | Tampering | ?next= open redirect | mitigate | Login page validates next param: must start with `/` and must not contain `://`. External URLs rejected, defaults to `/dashboard`. |
| T-02-19 | Spoofing | login form | mitigate | Form-level error "Invalid email or password" — does not reveal whether email exists (same error for wrong email and wrong password). Backend returns same 401 for both cases. |
| T-02-20 | Information Disclosure | signup 409 | accept | 409 on duplicate email reveals email existence. Accepted for v1 — standard UX pattern. Future: rate-limit signup endpoint (already done in Plan 02 via ThrottlerGuard). |
| T-02-21 | Tampering | profile form | mitigate | zod validation on frontend + class-validator on backend. Whitelist:true strips unknown fields. Only name and timezone are updatable. |
</threat_model>
